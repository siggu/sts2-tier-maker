local cargo = mw.ext.cargo
local Cards = require( 'Module:Cards' )
local cache = require 'mw.ext.LuaCache'
local p = {}

function getCargoData(name, sequel)
	local fields = 'Name, Image, Color, OverrideColor, Type, Rarity, Cost, CostPlus, StarCost, NoUpgrade, Description'
    local query_results = cargo.query('Cards', fields, { where = 'Cards.Name = "' .. name .. '" AND Cards.Game = "' .. sequel .. '"' })
    local data = query_results[1] or -1
    if data ~= -1 then
	    data.Cost = tonumber(data.Cost)
	    data.CostPlus = tonumber(data.CostPlus)
	    data.StarCost = tonumber(data.StarCost)
	    data.Description = Cards.combineDescription({args = {data.Description or ''}})
	end
    return data
end

function getCharacterLink(frame, color, sequel)
	local character_names = {
		["Red"] = "Ironclad",
		["Green"] = "Silent",
		["Blue"] = "Defect",
		["Purple"] = "Watcher",
		["Colorless"] = "Colorless"
	}
	
	return frame:preprocess("{{QueryLink|Cards|color:"..color.."|"..(character_names[color] or color).."|"..sequel.."|true}}")
end


function p.render(frame)
	local name = mw.text.trim(tostring(frame.args[1]))
	local sequel = frame.args[2] or "1"
	local cacheKey = name..sequel..'cardTooltip'
	
	local html = cache.get(cacheKey)

	if not html then
		local card = getCargoData(name, sequel)
		local actual_color = card.OverrideColor or card.Color
		
		local image = Cards.getIcon(name, actual_color, card.Rarity, card.Type, sequel)
		local type_text = ""
		if card.Type == "Curse" then
			type_text = "{{QueryLink|Cards|type:Curse|Curse|2|true}}"
		elseif card.Type == "Status" then
			type_text = "{{QueryLink|Cards|type:Status|Status|2|true}}"
		elseif card.Type == "Quest" then
			type_text = "{{QueryLink|Cards|type:Quest|Quest|2|true}}"
		else
			type_text = "{{QueryLink|Cards|rarity:"..card.Rarity.."&type:"..card.Type.."|"..card.Rarity.." "..card.Type.."|"..sequel.."|true}}"
		end
	
		html = mw.html.create('div'):addClass('tooltip-container')
	
		local main_section = html:tag('div'):css('padding', '5px')
		-- Header with card image, name, rarity, and class (if specified)
		local header = main_section:tag('div'):addClass('tooltip-header'):css('border-bottom', '1px solid gray')
		-- if card.Image then
		-- 	header:wikitext(image)
		-- end
		local labels = header:tag('div'):addClass('tooltip-header-labels'):css('gap', '4px'):css('margin-left', '0px')
		labels:wikitext('<span class=tooltip-title>' .. image .. " " .. name .. '</span>'):css('padding-bottom', '2px')
		if card.Rarity then
			labels:tag('span')
				:wikitext((card.Cost >= 0 and (card.Cost..(card.CostPlus and frame:preprocess("({{TS|positive||"..card.CostPlus.."}})") or "").." Energy") or card.Cost == -1 and "X Energy" or "Unplayable") ..
					(card.StarCost and card.StarCost > 0 and(" - "..card.StarCost .. "[[File:StS2_StarRegent.png|18px|middle]]") or card.StarCost and card.StarCost == -1 and " - X [[File:StS2_StarRegent.png|18px|middle]]" or ""))
				:wikitext(" - "..frame:preprocess(type_text))
				:wikitext(" - "..getCharacterLink(frame, card.Color, sequel))
				:addClass('tooltip-meta'):done()
		end
		labels:done()
		header:done()
		-- card Description
		main_section:tag('div'):addClass('tooltip-text'):wikitext(card.Description or ""):css('padding-top', '5px'):done()
		
		local art = html:tag('div'):addClass('card-tooltip-art')
		art:tag('span')
			:wikitext("[[File:"..card.Image:match("(.+)%..+$").."-Art.png|x100px|class=card-tooltip-art-image]]"):done()
	end
	
	cache.set(cacheKey, html)
	return tostring(html)
end

return p
